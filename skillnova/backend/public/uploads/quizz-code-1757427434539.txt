  function getQuestionBank() {
    const custom = readCustomSubjects();
    // Merge the default BANK with custom subjects
    const merged = {...BANK};
    
    // Mark default questions
    for (const key in BANK) {
      merged[key] = BANK[key].map(q => ({...q, isDefault: true}));
    }
    
    // Add custom questions without marking as default
    for (const key in custom) {
      if (!merged[key]) {
        merged[key] = [];
      }
      merged[key] = merged[key].concat(custom[key].map(q => ({...q, isDefault: false})));
    }
    
    return merged;
  }

    const today = new Date().toDateString();

    // Calculate statistics
    users.forEach(user => {
      if (user.history && user.history.length > 0) {
        totalQuizzes += user.history.length;
        
        user.history.forEach(attempt => {
          totalScore += attempt.score;
          scoreCount += attempt.items.length;
          
          if (new Date(attempt.dateISO).toDateString() === today) {
            activeToday++;
          }
        });
      }
    });
    
    $('totalQuizzes').textContent = totalQuizzes;
    $('avgScore').textContent = scoreCount > 0 ? Math.round((totalScore / scoreCount) * 100) + '%' : '0%';

  function renderAdminActivityChart() {
    const svg = $('adminActivityChart');
    svg.innerHTML = '';
    
    // Get activity data for the last 7 days
    const activityData = [];
    const today = new Date();
    
    for (let i = 6; i >= 0; i--) {
      const date = new Date(today);
      date.setDate(date.getDate() - i);
      const dateStr = date.toDateString();
      
      let quizzesCompleted = 0;
      users.forEach(user => {
        if (user.history) {
          user.history.forEach(attempt => {
            if (new Date(attempt.dateISO).toDateString() === dateStr) {
              quizzesCompleted++;
            }
          });
        }
      });
      
      activityData.push({
        date: date.toLocaleDateString('en-US', { weekday: 'short' }),
        quizzes: quizzesCompleted
      });
    }
    
    // Find max value for scaling
    const maxQuizzes = Math.max(...activityData.map(d => d.quizzes), 1);
    
    // Set up SVG dimensions
    const width = svg.clientWidth;
    const height = svg.clientHeight;
    const margin = { top: 20, right: 20, bottom: 30, left: 40 };
    const chartWidth = width - margin.left - margin.right;
    const chartHeight = height - margin.top - margin.bottom;
    
    // Create chart group
    const chart = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    chart.setAttribute('transform', `translate(${margin.left},${margin.top})`);
    svg.appendChild(chart);
    
    // Create X axis
    const xAxis = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    xAxis.setAttribute('transform', `translate(0,${chartHeight})`);
    chart.appendChild(xAxis);
    
    // Create Y axis
    const yAxis = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    chart.appendChild(yAxis);
    
    // Create bars
    const barWidth = chartWidth / activityData.length * 0.6;
    const barSpacing = chartWidth / activityData.length;
    
    activityData.forEach((data, i) => {
      const barHeight = (data.quizzes / maxQuizzes) * chartHeight;
      const x = i * barSpacing + (barSpacing - barWidth) / 2;
      const y = chartHeight - barHeight;
      
      // Create bar
      const bar = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
      bar.setAttribute('x', x);
      bar.setAttribute('y', y);
      bar.setAttribute('width', barWidth);
      bar.setAttribute('height', barHeight);
      bar.setAttribute('fill', '#1a73e8');
      bar.setAttribute('rx', '4');
      bar.style.transition = 'all 0.3s ease';
      
      // Add hover effect
      bar.addEventListener('mouseenter', function() {
        this.setAttribute('fill', '#1765cc');
        
        // Create tooltip
        const tooltip = document.createElementNS('http://www.w3.org/2000/svg', 'g');
        tooltip.setAttribute('id', 'tooltip');
        
        const tooltipRect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
        tooltipRect.setAttribute('x', x + barWidth / 2 - 30);
        tooltipRect.setAttribute('y', y - 30);
        tooltipRect.setAttribute('width', '60');
        tooltipRect.setAttribute('height', '20');
        tooltipRect.setAttribute('fill', 'rgba(0,0,0,0.8)');
        tooltipRect.setAttribute('rx', '4');
        
        const tooltipText = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        tooltipText.setAttribute('x', x + barWidth / 2);
        tooltipText.setAttribute('y', y - 15);
        tooltipText.setAttribute('text-anchor', 'middle');
        tooltipText.setAttribute('fill', 'white');
        tooltipText.setAttribute('font-size', '12');
        tooltipText.textContent = `${data.quizzes} quizzes`;
        
        tooltip.appendChild(tooltipRect);
        tooltip.appendChild(tooltipText);
        chart.appendChild(tooltip);
      });
      
      bar.addEventListener('mouseleave', function() {
        this.setAttribute('fill', '#1a73e8');
        const tooltip = document.getElementById('tooltip');
        if (tooltip) tooltip.remove();
      });
      
      chart.appendChild(bar);
      
      // Add X axis label
      const xLabel = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      xLabel.setAttribute('x', x + barWidth / 2);
      xLabel.setAttribute('y', chartHeight + 20);
      xLabel.setAttribute('text-anchor', 'middle');
      xLabel.setAttribute('font-size', '12');
      xLabel.textContent = data.date;
      chart.appendChild(xLabel);
    });
    
    // Add Y axis labels
    const yTicks = 5;
    for (let i = 0; i <= yTicks; i++) {
      const value = Math.round((i / yTicks) * maxQuizzes);
      const y = chartHeight - (i / yTicks) * chartHeight;
      
      const tick = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      tick.setAttribute('x1', -5);
      tick.setAttribute('y1', y);
      tick.setAttribute('x2', 0);
      tick.setAttribute('y2', y);
      tick.setAttribute('stroke', 'currentColor');
      tick.setAttribute('stroke-width', '1');
      yAxis.appendChild(tick);
      
      const label = document.createElementNS('http://www.w3.org/2000/svg', 'text');
      label.setAttribute('x', -10);
      label.setAttribute('y', y + 4);
      label.setAttribute('text-anchor', 'end');
      label.setAttribute('font-size', '12');
      label.textContent = value;
      yAxis.appendChild(label);
    }
    
    // Update legend
    const legend = $('adminChartLegend');
    legend.innerHTML = `
      <div class="performance-legend-item">
        <div class="performance-legend-color" style="background-color: #1a73e8"></div>
        <span>Quizzes Completed</span>
      </div>
    `;
  }

    Object.keys(bank).forEach(subject => {
      subjectStats[subject] = {
        totalQuizzes: 0,
        totalScore: 0,
        totalQuestions: 0
      };
    });
    
    // Calculate stats from user history
    users.forEach(user => {
      if (user.history) {
        user.history.forEach(attempt => {
          if (subjectStats[attempt.subject]) {
            subjectStats[attempt.subject].totalQuizzes++;
            subjectStats[attempt.subject].totalScore += attempt.score;
            subjectStats[attempt.subject].totalQuestions += attempt.items.length;
          }
        });
      }
    });
    
    // Generate colors for each subject
    const colors = ['#1a73e8', '#0f9d58', '#ea4335', '#fbbc04', '#4285f4', '#9c27b0', '#ff9800'];
    let colorIndex = 0;
    
    // Calculate total for percentages
    const totalQuizzes = Object.values(subjectStats).reduce((sum, stat) => sum + stat.totalQuizzes, 0);
    
    if (totalQuizzes === 0) {
      chartContainer.innerHTML = '<div style="text-align:center;padding:10px;">No quiz data available</div>';
      return;
    }
    
    // Create chart segments
    Object.entries(subjectStats).forEach(([subject, stats]) => {
      if (stats.totalQuizzes > 0) {
        const percentage = (stats.totalQuizzes / totalQuizzes) * 100;
        const avgScore = Math.round((stats.totalScore / stats.totalQuestions) * 100);
        
        const segment = document.createElement('div');
        segment.className = 'subject-chart-segment';
        segment.style.width = `${percentage}%`;
        segment.style.backgroundColor = colors[colorIndex % colors.length];
        segment.title = `${subject}: ${stats.totalQuizzes} quizzes (${percentage.toFixed(1)}%), Avg: ${avgScore}%`;
        
        chartContainer.appendChild(segment);
        
        // Add legend item
        const legendItem = document.createElement('div');
        legendItem.className = 'subject-legend-item';
        legendItem.innerHTML = `
          <div class="subject-legend-color" style="background-color: ${colors[colorIndex % colors.length]}"></div>
          <span>${subject} (${stats.totalQuizzes})</span>
        `;
        
        legendContainer.appendChild(legendItem);
        
        colorIndex++;
      }
    });
  }

  function openUserDrawer(user){ 
    $('drawerTitle').textContent = `${user.username} — History`; 
    $('drawerMeta').textContent = user.email||''; 
    const content=$('drawerContent'); 
    content.innerHTML=''; 
    
    const hist=user.history||[]; 
    
    if(hist.length===0){
      content.innerHTML=`<div class='small'>No attempts yet.</div>`; 
    } else { 
      // Group by subject
      const subjectGroups = {};
      hist.forEach(attempt => {
        if (!subjectGroups[attempt.subject]) {
          subjectGroups[attempt.subject] = [];
        }
        subjectGroups[attempt.subject].push(attempt);
      });
      
      // For each subject
      for (const subject in subjectGroups) {
        const attempts = subjectGroups[subject];
        // Calculate stats
        const scores = attempts.map(a => a.score);
        const maxQuestions = attempts[0].items.length;
        const bestScore = Math.max(...scores);
        const avgScore = scores.reduce((a, b) => a + b, 0) / scores.length;
        const lastAttempt = attempts.reduce((latest, current) => 
          new Date(current.dateISO) > new Date(latest.dateISO) ? current : latest
        );
        // Create subject section
        const subjectDiv = document.createElement('div');
        subjectDiv.style.marginBottom = '24px';
        subjectDiv.innerHTML = `
          <h3 style="margin:0 0 8px 0; font-size:18px;">${subject}</h3>
          <div style="margin-bottom:12px; font-size:14px; color:var(--muted);">
            Best: ${bestScore}/${maxQuestions}, Avg: ${Math.round(avgScore)}, Attempts: ${attempts.length}, Last: ${new Date(lastAttempt.dateISO).toLocaleDateString()}
          </div>
        `;
        // Create attempts list for this subject
        const attemptsList = document.createElement('div');
        attemptsList.style.marginLeft = '16px';
        attempts.forEach((attempt, index) => {
          const attemptDiv = document.createElement('div');

  function startQuiz(subject){ 
    if(!current){ 
      alert('Please login first.'); 
      showView('viewLogin'); 
      return; 
    } 
    
    lastQuizSubject = subject;
    const bank = getQuestionBank();
    if (!bank[subject]) {
      alert('Subject not found');
      return;
    }
    
    const bankSubject = bank[subject];
    
    // Initialize quiz state with new features
    quiz = {
      subject, 
      items: bankSubject.map(q => ({...q})),
      currentIndex: 0,
      visited: new Array(bankSubject.length).fill(false),
      marked: new Array(bankSubject.length).fill(false),
      answers: new Array(bankSubject.length).fill(null),
      timeLeft: 600, // 10 minutes in seconds
      startTime: Date.now(),
      captured: []
    };
    
    // Mark first question as visited
    quiz.visited[0] = true;
    
    // Update UI
    $('chipSubject').textContent = subject.toUpperCase();
    $('totalQuestions').textContent = bankSubject.length;
    updateQuestionPalette();
    updateQuizStats();
    renderQuestion();
    startTimer();
    showView('viewQuiz');
  }
  
  function updateQuestionPalette() {
    const palette = $('questionPalette');
    palette.innerHTML = '';
    
    for (let i = 0; i < quiz.items.length; i++) {
      const btn = document.createElement('button');
      btn.className = 'question-btn';
      btn.textContent = i + 1;
      btn.dataset.index = i;
      
      if (i === quiz.currentIndex) {
        btn.classList.add('current');
      } else if (quiz.answers[i] !== null) {
        btn.classList.add('answered');
      } else if (quiz.marked[i]) {
        btn.classList.add('marked');
      } else if (quiz.visited[i]) {
        btn.style.background = 'rgba(26,115,232,0.08)';
      }
      
      btn.addEventListener('click', () => {
        // Save current question state before navigating
        saveCurrentQuestionState();
        
        // Navigate to selected question
        quiz.currentIndex = i;
        quiz.visited[i] = true;
        renderQuestion();
        updateQuestionPalette();
      });
      
      palette.appendChild(btn);
    }
  }
  
  function updateQuizStats() {
    const answeredCount = quiz.answers.filter(a => a !== null).length;
    $('currentQuestion').textContent = `Question ${quiz.currentIndex + 1}`;
    $('answeredCount').textContent = `Answered: ${answeredCount}`;
    
    // Update progress bar
    const progress = (answeredCount / quiz.items.length) * 100;
    $('barFill').style.width = `${progress}%`;
  }
  
  function renderQuestion() {
    if (!quiz) return;
    
    const cur = quiz.items[quiz.currentIndex];
    $('qText').textContent = `Q${quiz.currentIndex + 1}. ${cur.q}`;
    
    const opts = $('opts');
    opts.innerHTML = '';
    
    cur.options.forEach((opt, idx) => {
      const d = document.createElement('div');
      d.className = 'option';
      d.tabIndex = 0;
      d.setAttribute('role', 'button');
      d.setAttribute('aria-pressed', 'false');
      d.textContent = opt;
      
      // Check if this option was previously selected
      if (quiz.answers[quiz.currentIndex] === idx) {
        d.style.background = 'rgba(26,115,232,0.16)';
        d.style.borderColor = 'var(--accent-solid)';
      }
      
      d.addEventListener('click', () => selectOption(idx, d));
      d.addEventListener('keydown', e => {
        if (e.key === 'Enter') selectOption(idx, d);
      });
      opts.appendChild(d);
    });
    
    // Update question palette to highlight current question
    updateQuestionPalette();
    updateQuizStats();
    
    // Update mark review button state
    $('btnMarkReview').textContent = quiz.marked[quiz.currentIndex] ? 'Unmark Review' : 'Mark for Review';
  }
  
  function selectOption(idx, element) {
    if (!quiz) return;
    
    // Update the answer
    quiz.answers[quiz.currentIndex] = idx;
    
    // Update UI to show selected option
    document.querySelectorAll('.option').forEach(opt => {
      opt.style.background = '';
      opt.style.borderColor = '';
    });
    
    element.style.background = 'rgba(26,115,232,0.16)';
    element.style.borderColor = 'var(--accent-solid)';
    
    // Update stats
    updateQuizStats();
    updateQuestionPalette();
  }

  function startTimer() {
    if (timerId) clearInterval(timerId);
    
    timerId = setInterval(() => {
      quiz.timeLeft--;
      
      // Format time as MM:SS
      const minutes = Math.floor(quiz.timeLeft / 60);
      const seconds = quiz.timeLeft % 60;
      $('timer').textContent = `${minutes}:${seconds < 10 ? '0' : ''}${seconds}`;
      
      if (quiz.timeLeft <= 0) {
        clearInterval(timerId);
        timerId = null;
        finishQuiz();
      }
    }, 1000);
  }
  
  function navigateQuestion(direction) {
    if (!quiz) return;
    
    saveCurrentQuestionState();
    
    if (direction === 'next' && quiz.currentIndex < quiz.items.length - 1) {
      quiz.currentIndex++;
    } else if (direction === 'prev' && quiz.currentIndex > 0) {
      quiz.currentIndex--;
    }
    
    quiz.visited[quiz.currentIndex] = true;
    renderQuestion();
  }

  function finishQuiz() {
    if (timerId) {
      clearInterval(timerId);
      timerId = null;
    }
    
    if (!quiz) return;
    
    const timeUsed = 600 - quiz.timeLeft; // 10 minutes - time left
    const answeredCount = quiz.answers.filter(a => a !== null).length;
    const skippedCount = quiz.items.length - answeredCount;
    
    let correct = 0;
    let wrong = 0;
    
    // Process each question
    for (let i = 0; i < quiz.items.length; i++) {
      const question = quiz.items[i];
      const userAnswer = quiz.answers[i];
      
      // Capture question details for results
      quiz.captured.push({
        q: question.q,
        options: question.options.slice(),
        selectedIndex: userAnswer,
        correctIndex: question.answerIndex,
        marked: quiz.marked[i]
      });
      
      // Count correct and wrong answers
      if (userAnswer !== null) {
        if (userAnswer === question.answerIndex) {
          correct++;
        } else {
          wrong++;
        }
      }
    }
    
    const attempt = {
      subject: quiz.subject,
      score: correct,
      correct,
      wrong,
      skipped: skippedCount,
      timeUsed,
      dateISO: new Date().toISOString(),
      items: quiz.captured
    };
    
    // Store the last quiz result for comparison
    lastQuizResult = attempt;
    
    users = readUsers();
    const idxUser = users.findIndex(u => u.username === current.username);
    
    if (idxUser !== -1) {
      users[idxUser].history = users[idxUser].history || [];
      users[idxUser].history.push(attempt);
      writeUsers(users);
    }
    
    // Update result UI
    $('resultSubjectDate').textContent = `${quiz.subject.charAt(0).toUpperCase() + quiz.subject.slice(1)} • ${new Date(attempt.dateISO).toLocaleDateString()}`;
    $('finalScore').textContent = `${attempt.score} / ${quiz.items.length}`;
    $('resCorrect').textContent = attempt.correct;
    $('resWrong').textContent = attempt.wrong;
    $('resSkipped').textContent = attempt.skipped;
    $('resTime').textContent = `${Math.floor(attempt.timeUsed / 60)}:${(attempt.timeUsed % 60).toString().padStart(2, '0')}`;
    
    // Calculate percentages for progress chart
    const total = quiz.items.length;
    const correctPercent = Math.round((attempt.correct / total) * 100);
    const wrongPercent = Math.round((attempt.wrong / total) * 100);
    const skippedPercent = Math.round((attempt.skipped / total) * 100);
    
    // Update progress chart
    $('correctSegment').style.width = `${correctPercent}%`;
    $('correctSegment').textContent = correctPercent > 10 ? `${correctPercent}%` : '';
    $('wrongSegment').style.width = `${wrongPercent}%`;
    $('wrongSegment').textContent = wrongPercent > 10 ? `${wrongPercent}%` : '';
    $('skippedSegment').style.width = `${skippedPercent}%`;
    $('skippedSegment').textContent = skippedPercent > 10 ? `${skippedPercent}%` : '';
    
    // Calculate performance percentage
    const percentage = Math.round((attempt.score / quiz.items.length) * 100);
    const performanceIndicator = $('performanceIndicator');
    
    // Clear previous classes
    performanceIndicator.className = 'performance-indicator';
    
    // Set performance message and styling based on score
    if (percentage >= 80) {
      performanceIndicator.textContent = 'Excellent Work! Keep it up!';
      performanceIndicator.classList.add('performance-excellent');
    } else if (percentage >= 60) {
      performanceIndicator.textContent = 'Good Job! Room for improvement.';
      performanceIndicator.classList.add('performance-good');
    } else if (percentage >= 40) {
      performanceIndicator.textContent = 'Needs Improvement. Try again!';
      performanceIndicator.classList.add('performance-improvement');
    } else {
      performanceIndicator.textContent = 'Keep Practicing! You can do better.';
      performanceIndicator.classList.add('performance-poor');
    }
    
    // Populate question review
    const questionReview = $('questionReview');
    questionReview.innerHTML = '';
    
    quiz.captured.forEach((item, index) => {
      const isCorrect = item.selectedIndex === item.correctIndex;
      const chosen = (typeof item.selectedIndex === 'number') ? item.options[item.selectedIndex] : '—';
      const correct = item.options[item.correctIndex];
      
      const questionItem = document.createElement('div');
      questionItem.className = 'question-item';
      
      questionItem.innerHTML = `
        <div class="question-text">Q${index + 1}: ${item.q}</div>
        <div class="question-result">
          <span class="${isCorrect ? 'correct-answer' : 'wrong-answer'}">
            ${isCorrect ? 'Correct' : 'Incorrect'}
          </span>
          <span>Your answer: ${chosen}</span>
          <span>Correct answer: ${correct}</span>
        </div>
      `;
      
      questionReview.appendChild(questionItem);
    });
    
    // Show performance comparison if there are previous attempts
    if (users[idxUser] && users[idxUser].history && users[idxUser].history.length > 1) {
      showPerformanceComparison(users[idxUser].history, attempt);
    } else {
      $('resultComparison').style.display = 'none';
    }
    
    // Show insights
    showResultInsights(attempt, quiz.items.length);
    
    quiz = null;
    showView('viewResult');
  }

  function renderPerformanceChart(user) {
    const container = $('chartBarContainer');
    container.innerHTML = '';
    
    // Get subject performance data
    const subjectStats = {};
    
    // Initialize with all subjects
    const bank = getQuestionBank();
    Object.keys(bank).forEach(subject => {
      subjectStats[subject] = {
        bestScore: 0,
        totalQuestions: bank[subject].length,
        attempts: 0
      };
    });
    
    // Calculate stats from user history
    if (user.history) {
      user.history.forEach(attempt => {
        if (subjectStats[attempt.subject]) {
          subjectStats[attempt.subject].attempts++;
          subjectStats[attempt.subject].bestScore = Math.max(
            subjectStats[attempt.subject].bestScore,
            attempt.score
          );
        }
      });
    }
    
    // Generate colors for each subject
    const colors = ['#1a73e8', '#0f9d58', '#ea4335', '#fbbc04', '#4285f4', '#9c27b0', '#ff9800'];
    let colorIndex = 0;
    
    // Find max value for scaling
    const maxScore = Math.max(...Object.values(subjectStats).map(s => s.bestScore), 1);
    
    // Create chart bars
    Object.entries(subjectStats).forEach(([subject, stats]) => {
      if (stats.attempts > 0) {
        const percentage = (stats.bestScore / stats.totalQuestions) * 100;
        const barHeight = (stats.bestScore / maxScore) * 160; // Max height is 160px
        
        const barContainer = document.createElement('div');
        barContainer.className = 'chart-bar';
        
        const barFill = document.createElement('div');
        barFill.className = 'chart-bar-fill';
        barFill.style.height = `${barHeight}px`;
        barFill.style.backgroundColor = colors[colorIndex % colors.length];
        
        const barValue = document.createElement('div');
        barValue.className = 'chart-bar-value';
        barValue.textContent = `${Math.round(percentage)}%`;
        
        const barLabel = document.createElement('div');
        barLabel.className = 'chart-bar-label';
        barLabel.textContent = subject.charAt(0).toUpperCase() + subject.slice(1);
        
        barFill.appendChild(barValue);
        barContainer.appendChild(barFill);
        barContainer.appendChild(barLabel);
        container.appendChild(barContainer);
        
        colorIndex++;
      }
    });
    
    // Update legend
    const legend = $('chartLegend');
    legend.innerHTML = '';
    
    colorIndex = 0;
    Object.entries(subjectStats).forEach(([subject, stats]) => {
      if (stats.attempts > 0) {
        const legendItem = document.createElement('div');
        legendItem.className = 'chart-legend-item';
        legendItem.innerHTML = `

          <span>${subject.charAt(0).toUpperCase() + subject.slice(1)}</span>

        legend.appendChild(legendItem);

      }
    });

    const user = users.find(u => u.username === current.username);

    const allQuestions = [];
    
    user.history.forEach(attempt => {
      attempt.items.forEach((item, index) => {
        // Check if this question already exists in our collection
        const existingQuestionIndex = allQuestions.findIndex(q =>          q.text === item.q);
        
        if (existingQuestionIndex !== -1) {
          // Update existing question with additional attempt data
          allQuestions[existingQuestionIndex].attempts.push({
            date: attempt.dateISO,
            subject: attempt.subject,
            selectedIndex: item.selectedIndex,
            isCorrect: item.selectedIndex === item.correctIndex
          });
        } else {
          // Add new question
          allQuestions.push({
            text: item.q,
            options: item.options,
            correctIndex: item.correctIndex,
            subject: attempt.subject,
            attempts: [{
              date: attempt.dateISO,
              subject: attempt.subject,
              selectedIndex: item.selectedIndex,
              isCorrect: item.selectedIndex === item.correctIndex
            }]
          });
        }
      });
    });
    
    renderQuestionViewer(allQuestions);
    showView('viewQuestionViewer');
  }
  
  function renderQuestionViewer(questions) {
    const container = $('questionCategories');
    container.innerHTML = '';
    
    if (questions.length === 0) {
      container.innerHTML = '<div style="text-align:center;padding:40px;">No questions found</div>';
      return;
    }
    
    // Group questions by subject
    const questionsBySubject = {};
    questions.forEach(question => {
      if (!questionsBySubject[question.subject]) {
        questionsBySubject[question.subject] = [];
      }
      questionsBySubject[question.subject].push(question);
    });
    
    // Create category for each subject
    for (const subject in questionsBySubject) {
      const subjectQuestions = questionsBySubject[subject];
      
      // Calculate stats for this subject
      const totalAttempts = subjectQuestions.reduce((sum, q) => sum + q.attempts.length, 0);
      const correctAttempts = subjectQuestions.reduce((sum, q) => 
        sum + q.attempts.filter(a => a.isCorrect).length, 0);
      const accuracy = totalAttempts > 0 ? Math.round((correctAttempts / totalAttempts) * 100) : 0;
      
      // Create category section
      const categoryDiv = document.createElement('div');
      categoryDiv.className = 'question-category';
      
      categoryDiv.innerHTML = `
        <div class="question-category-header">
          <div>${subject.charAt(0).toUpperCase() + subject.slice(1)}</div>

  async function handleFile(file) {

  async function extractTextFromFile(file) {
    return new Promise((resolve, reject) => {
      const fileType = file.name.split('.').pop().toLowerCase();
      
      if (fileType === 'txt') {
        const reader = new FileReader();
        reader.onload = (e) => resolve(e.target.result);
        reader.onerror = reject;
        reader.readAsText(file);
      } else if (fileType === 'pdf') {
        const reader = new FileReader();
        reader.onload = function() {
          const typedarray = new Uint8Array(this.result);
          
          // Set the worker source for pdf.js
          pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.4.120/pdf.worker.min.js';
          
          pdfjsLib.getDocument(typedarray).promise.then(function(pdf) {
            let textContent = '';
            const numPages = pdf.numPages;
            let count = 0;
            
            for (let i = 1; i <= numPages; i++) {
              pdf.getPage(i).then(function(page) {
                return page.getTextContent();
              }).then(function(text) {
                return text.items.map(function(s) { return s.str; }).join(' ');
              }).then(function(text) {
                textContent += text + '\n';
                count++;
                if (count === numPages) {
                  resolve(textContent);
                }
              }).catch(reject);
            }
          }).catch(reject);
        };
        reader.readAsArrayBuffer(file);
      } else if (fileType === 'doc' || fileType === 'docx') {
        const reader = new FileReader();
        reader.onload = function(e) {
          const arrayBuffer = e.target.result;
          mammoth.extractRawText({ arrayBuffer: arrayBuffer })
            .then(function(result) {
              resolve(result.value);
            })
            .catch(reject);
        };
        reader.readAsArrayBuffer(file);
      } else {
        reject(new Error('Unsupported file type'));
      }
    });
  }

    }
  }
  
  function parseQuestions(text) {
  const lines = text.split('\n');
  const questions = [];
  let currentQuestion = null;
  let currentOptions = [];
  let currentAnswer = null;
  
  for (let line of lines) {
    line = line.trim();
    if (line === '') continue; // Skip empty lines
    
    // Check if it's a question (starts with number and period)
    if (/^\d+\.\s/.test(line)) {
      // If we have a current question being built, push it
      if (currentQuestion && currentOptions.length > 0) {
        questions.push({
          q: currentQuestion,
          options: currentOptions,
          answerIndex: currentAnswer || 0, // Default to first option if no answer found
          isDefault: false
        });
      }
      // Start new question
      currentQuestion = line.replace(/^\d+\.\s/, '');
      currentOptions = [];
      currentAnswer = null;
    } 
    // Check if it's an option (starts with letter and closing parenthesis)
    else if (/^[a-z]\)\s/.test(line)) {
      const optionText = line.replace(/^[a-z]\)\s/, '');
      currentOptions.push(optionText);
    }
    // Check if it's an answer line (if present)
    else if (line.startsWith('ANSWER:')) {
      const answerLetter = line.substring(7).trim().toLowerCase();
      const index = answerLetter.charCodeAt(0) - 'a'.charCodeAt(0);
      if (index >= 0 && index < currentOptions.length) {
        currentAnswer = index;
      }
    }
  }
  
  // Push the last question
  if (currentQuestion && currentOptions.length > 0) {
    questions.push({
      q: currentQuestion,
      options: currentOptions,
      answerIndex: currentAnswer || 0, // Default to first option if no answer found
      isDefault: false
    });
  }
  
  return questions;
}

  function deleteSubjectQuestion(subject, questionIndex) {
    const bank = getQuestionBank();
    const question = bank[subject][questionIndex];
    
    if (question.isDefault) {
      alert('Default questions cannot be deleted.');
      return;
    }
    
    if (!confirm('Are you sure you want to delete this question?')) return;
    
    const customSubjects = readCustomSubjects();
    
    if (customSubjects[subject]) {
      // Find the index in the custom subjects (not the merged bank)
      // We need to find the question in the custom subjects that matches the one we want to delete
      const customQuestions = customSubjects[subject];
      const questionToDelete = customQuestions.find(q => 
        q.q === question.q && 
        q.options.join(',') === question.options.join(',') && 
        q.answerIndex === question.answerIndex
      );
      
      if (questionToDelete) {
        const index = customQuestions.indexOf(questionToDelete);
        customQuestions.splice(index, 1);
        writeCustomSubjects(customSubjects);
        
        // Refresh the questions view
        const bank = getQuestionBank();
        const questions = bank[subject] || [];
        renderSubjectQuestions(questions);
      }
    }
  }

    const customSubjects = readCustomSubjects();
    
    if (!customSubjects[currentSubjectForQuestions]) {
      customSubjects[currentSubjectForQuestions] = [];
    }
    
    if (editingQuestionIndex !== null) {
      // Update existing question
      customSubjects[currentSubjectForQuestions][editingQuestionIndex] = question;
    } else {
      // Add new question
      customSubjects[currentSubjectForQuestions].push(question);
    }
    
    writeCustomSubjects(customSubjects);
    
    // Refresh the questions view
    const bank = getQuestionBank();
    const questions = bank[currentSubjectForQuestions] || [];
    renderSubjectQuestions(questions);
    
    // Go back to questions view
    showView('viewSubjectQuestions');
  }
  
  function addNewOption() {
    const optionsContainer = $('optionsContainer');
    const optionCount = optionsContainer.querySelectorAll('.option-item').length;
    
    if (optionCount >= 6) {
      alert('Maximum 6 options allowed');
      return;
    }
    
    const optionDiv = document.createElement('div');
    optionDiv.className = 'option-item';
    optionDiv.innerHTML = `
      <input type="text" class="form-input option-input" placeholder="Option ${String.fromCharCode(65 + optionCount)}">
      <button type="button" class="option-remove">✕</button>
    `;
    
    optionsContainer.appendChild(optionDiv);
    
    // Add event listener to remove button
    optionDiv.querySelector('.option-remove').addEventListener('click', function() {
      optionsContainer.removeChild(optionDiv);
      updateCorrectAnswerOptions();
    });
    
    // Update correct answer options
    updateCorrectAnswerOptions();
  }
  
  function updateCorrectAnswerOptions() {
    const optionsContainer = $('optionsContainer');

    correctOptionsContainer.innerHTML = '';

  function renderFilteredUsers(filteredUsers) {
    const grid = $('usersGrid'); 
    grid.innerHTML=''; 
    
    if(filteredUsers.length===0){
      const e=document.createElement('div'); 
      e.className='card'; 
      e.innerHTML=`
        <div class="title">No users match the filter</div>
        <div class="meta">Try a different filter</div>
      `; 
      grid.appendChild(e);
      return; 
    }
    
    filteredUsers.forEach(u=>{ 
      const c=document.createElement('div'); 
      c.className='card'; 
      const subjects=[...new Set((u.history||[]).map(h=>h.subject))].join(', ')||'—'; 
      const totalAttempts = (u.history || []).length;
      const avgScore = totalAttempts > 0 ? 
        Math.round((u.history.reduce((sum, h) => sum + h.score, 0) / totalAttempts)) : 0;
      
      c.innerHTML=`
        <div class="title">${u.username}</div>
        <div class="meta">${u.email||''}</div>
        <div style='margin-top:12px' class='small'>Subjects: ${subjects}</div>
        <div style='margin-top:8px' class='small'>Attempts: ${totalAttempts} | Avg Score: ${avgScore}%</div>
        <div style='margin-top:12px;display:flex;gap:6px'>
          <button class='btn ghost small' data-user='${u.username}' data-action='view'>View Details</button>
          <button class='btn ghost small' data-user='${u.username}' data-action='del'>Delete</button>
        </div>
      `; 
      
      c.querySelectorAll('button').forEach(btn=>{ 
        btn.addEventListener('click', adminUserAction); 
      }); 
      
      grid.appendChild(c); 
    }); 
  }
